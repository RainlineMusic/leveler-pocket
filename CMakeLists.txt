cmake_minimum_required(VERSION 3.22)
project(LevelPocket VERSION 0.2.0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LEVEL_DSP_ONLY "Build portable DSP tests without JUCE" OFF)
option(LEVEL_ENABLE_OPENGL "Keep optional OpenGL component renderer" ON)
option(LEVEL_BUILD_UI_TESTS "Build processor and editor integration checks" ON)
set(LEVEL_FORMATS "VST3;AAX;Standalone" CACHE STRING "Plugin formats")
set(LEVEL_SANITIZER "none" CACHE STRING "none, address, undefined")
if(NOT LEVEL_SANITIZER STREQUAL "none" AND NOT MSVC)
 if(LEVEL_SANITIZER STREQUAL "address")
  set(SAN_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
 elseif(LEVEL_SANITIZER STREQUAL "undefined")
  set(SAN_FLAGS -fsanitize=undefined -fno-omit-frame-pointer)
 else()
  message(FATAL_ERROR "Unsupported sanitizer")
 endif()
 add_compile_options(${SAN_FLAGS})
 add_link_options(${SAN_FLAGS})
endif()
enable_testing()
add_executable(LevelDSPTest Tests/dsp_test.cpp)
add_test(NAME LevelDSP COMMAND LevelDSPTest)
if(LEVEL_DSP_ONLY)
 return()
endif()
if(NOT DEFINED JUCE_DIR)
 set(JUCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../JUCE")
endif()
if(WIN32)
 include(cmake/ApplyWindowsVBlankFix.cmake)
 duck_apply_windows_vblank_fix("${JUCE_DIR}")
endif()
add_subdirectory(${JUCE_DIR} JUCE)
juce_add_plugin(LevelPocket COMPANY_NAME "Rainline Music" COMPANY_WEBSITE "https://github.com/RainlineMusic" COMPANY_EMAIL "rainlinemusic@gmail.com" BUNDLE_ID "ru.rainlinemusic.levelpocket" AAX_IDENTIFIER "ru.rainlinemusic.levelpocket" PLUGIN_MANUFACTURER_CODE RnLn PLUGIN_CODE LvPk FORMATS ${LEVEL_FORMATS} PRODUCT_NAME "Level Pocket" DESCRIPTION "Broadband vocal compressor with adaptive spectral detector" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics AAX_CATEGORY Dynamics)
juce_generate_juce_header(LevelPocket)
target_sources(LevelPocket PRIVATE Source/LevelDSP.h Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h Source/UIStyle.h)
target_compile_definitions(LevelPocket PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 LEVEL_ENABLE_OPENGL=$<BOOL:${LEVEL_ENABLE_OPENGL}> $<$<BOOL:${WIN32}>:NOMINMAX=1> $<$<BOOL:${WIN32}>:WIN32_LEAN_AND_MEAN=1>)
if(MSVC)
 target_compile_options(LevelPocket PRIVATE /utf-8)
else()
 target_compile_options(LevelPocket PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
target_link_libraries(LevelPocket PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
if(LEVEL_ENABLE_OPENGL)
 target_link_libraries(LevelPocket PUBLIC juce::juce_opengl)
endif()
juce_add_binary_data(LevelAssets SOURCES Assets/Fonts/Inter-Regular.ttf Assets/Fonts/Inter-Medium.ttf Assets/Designer/DialLarge.png Assets/Designer/DialSmall.png)
set_target_properties(LevelAssets PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
target_link_libraries(LevelPocket PUBLIC LevelAssets)
foreach(format VST3 AAX)
 if(TARGET LevelPocket_${format})
  add_custom_command(TARGET LevelPocket_${format} POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:LevelPocket_${format}>/../Resources" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "$<TARGET_FILE_DIR:LevelPocket_${format}>/../Resources/Inter-OFL.txt" VERBATIM)
 endif()
endforeach()
if(LEVEL_BUILD_UI_TESTS)
 juce_add_console_app(LevelIntegrationTest PRODUCT_NAME "Level Pocket integration tests")
 juce_generate_juce_header(LevelIntegrationTest)
 target_sources(LevelIntegrationTest PRIVATE Tests/integration_test.cpp)
 target_include_directories(LevelIntegrationTest PRIVATE Source ${CMAKE_CURRENT_BINARY_DIR}/LevelPocket_artefacts/JuceLibraryCode)
 target_link_libraries(LevelIntegrationTest PRIVATE LevelPocket juce::juce_audio_utils)
 add_test(NAME LevelIntegration COMMAND LevelIntegrationTest)
endif()
